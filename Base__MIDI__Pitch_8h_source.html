143 //Operates device per desired MIDI output
144 inline void processNotes()
145 {
146 if(_currentNote < 0)
147 {
148 _debug.debugln(ISR, F("%d - No note"), _id);
149 return;
150 }
151
152 if(_currentNote == 0) //Reset pending
153 {
154 _debug.debugln(ISR, F("%d - Resetting properties"), _id);
155 resetProperties();
156 return;
157 }
158
159 _currentTick++;
160 if(_currentTick >= _currentPeriod) //Update pending
161 {
162 _debug.debugln(ISR, F("%d - Cycling output"), _id);
163 cycleOutput();
164 _currentTick = 0; //Reset counter
165 }
166
167 _currentDuration.addMicros(MIDI_Periods::getResolution());
168
169 if(_currentEffect == Vibrato)
170 _vibratoTick += MIDI_Periods::getResolution();
171 };

217 inline void processVibrato()
218 {
219 if(_currentEffect != Vibrato)
220 return; //Vibrato not being applied
221
222 if(_vibratoTick < _vibratoRate)
223 return; //Not yet time to increment
224
225 if(_vibratoDegree > 359)
226 _vibratoDegree = 0;
227
228 int16_t b = _vibratoAmount * SinWave::getSinValue(_vibratoDegree++);
229 _currentPeriod = getBasePeriod() / PitchBend::calculateFactor(b);
230 _vibratoTick = 0;
231 }

239 inline void bendNote(int16_t bend, bool shiftRange = false)
240 {
241 if (!(_currentNote > 0 && _currentNote < 256))

243 _debug.println(F("%d - Not bending; no note assigned"), _id);

249 {
250 _debug.println(F("%d - Not bending; processing some other effect"), _id);

253
254 //Set the current effect so it's not interfered with
255 //Or clear it if the bend is 0 (bend back to original note)
256 _currentEffect = bend == 0 ? None : PitchBend;
257
258 float pitchFactor = PitchBend::calculateFactor(bend, shiftRange);
259 uint32_t basePeriod = getBasePeriod();
260 uint32_t newPeriod = basePeriod / pitchFactor;
261 _currentPeriod = newPeriod;
262 _debug.debugln(DEBUG, F("%d - Bending note %d by %d (%f) from %d to %d"), _id, _currentNote, bend, pitchFactor, basePeriod, newPeriod);
263 };
