167 inline void manualIOUpdate()
168 {
169 //Call updateOutputs on unique IOs
170 uint8_t updateCount = 0;
171 IO_Device* tmpIOs[] = { _outIO1, _outIO2, _outIO3, _outIO4 };
172 for(int x = 0; x < 4; x++)
173 {
174 if(!tmpIOs[x]) continue; //No IO set
175 callUpdateOutputs(tmpIOs[x]);
176 updateCount++;
177
178 //Remove duplicates from the rest of the array
179 for(int y = (x + 1); y < 4; y++)
180 {
181 if(!tmpIOs[y]) continue; //No IO set
182 if(tmpIOs[x] == tmpIOs[y]) tmpIOs[y] = nullptr; //Same IO, remove
183 }
184 }
185
186 _debug.debugln(DEBUG, F("Updated %d IO(s)"), updateCount);
187 }
